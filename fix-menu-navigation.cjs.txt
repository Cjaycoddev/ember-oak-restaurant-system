const fs = require("fs");
const path = require("path");

const root = process.cwd();

const menuFile = path.join(
  root,
  "src",
  "features",
  "menu",
  "pages",
  "MenuPage.jsx"
);

const featuredFile = path.join(
  root,
  "src",
  "features",
  "home",
  "sections",
  "FeaturedDishes.jsx"
);

const specialsFile = path.join(
  root,
  "src",
  "features",
  "home",
  "sections",
  "ChefSpecials.jsx"
);

const cssFile = path.join(root, "src", "index.css");

function read(file) {
  if (!fs.existsSync(file)) {
    throw new Error(`File not found: ${file}`);
  }

  return fs.readFileSync(file, "utf8");
}

function write(file, content) {
  fs.writeFileSync(file, content, "utf8");
}

/* =====================================================
   MENU PAGE
===================================================== */

let menu = read(menuFile);

/* Add useLocation + useEffect */
if (!menu.includes('from "react-router-dom"')) {
  menu = menu.replace(
    'import { useMemo, useState } from "react";',
    'import { useEffect, useMemo, useState } from "react";\nimport { useLocation } from "react-router-dom";'
  );
} else {
  throw new Error(
    "Unexpected react-router-dom import already exists in MenuPage.jsx. Stopping to avoid damaging the file."
  );
}

/* Add location + highlight state */
const stateBlock = `  const [activeCategory, setActiveCategory] = useState("all");
  const [searchInput, setSearchInput] = useState("");
  const [searchTerm, setSearchTerm] = useState("");`;

const newStateBlock = `  const location = useLocation();

  const [activeCategory, setActiveCategory] = useState("all");
  const [searchInput, setSearchInput] = useState("");
  const [searchTerm, setSearchTerm] = useState("");
  const [highlightedDish, setHighlightedDish] = useState("");`;

if (!menu.includes(newStateBlock)) {
  if (!menu.includes(stateBlock)) {
    throw new Error("Could not find MenuPage state block.");
  }

  menu = menu.replace(stateBlock, newStateBlock);
}

/* Add reliable hash navigation effect */
if (!menu.includes("handleDishHash")) {
  const marker = `  const activeSection = useMemo(`;

  const effect = `  useEffect(() => {
    const dishId = location.hash.replace("#", "");

    if (!dishId.startsWith("dish-")) {
      return;
    }

    const timer = window.setTimeout(() => {
      const element = document.getElementById(dishId);

      if (!element) {
        console.warn(\`Ember & Oak: dish element not found: \${dishId}\`);
        return;
      }

      element.scrollIntoView({
        behavior: "smooth",
        block: "center",
      });

      setHighlightedDish(dishId);

      window.setTimeout(() => {
        setHighlightedDish("");
      }, 3000);
    }, 150);

    return () => {
      window.clearTimeout(timer);
    };
  }, [location.hash]);

`;

  if (!menu.includes(marker)) {
    throw new Error("Could not find activeSection block.");
  }

  menu = menu.replace(marker, effect + marker);
}

/* Replace MenuCard */
const oldMenuCard = `function MenuCard({ item }) {
  return (
    <article className="group flex min-w-0 flex-col overflow-hidden border border-zinc-800 bg-[#151412] transition-all duration-300 hover:-translate-y-1 hover:border-amber-400/70 hover:bg-[#191714]">`;

const newMenuCard = `function MenuCard({ item, dishId, highlighted = false }) {
  return (
    <article
      id={dishId}
      className={[
        "group flex min-w-0 flex-col overflow-hidden border border-zinc-800 bg-[#151412] transition-all duration-300 hover:-translate-y-1 hover:border-amber-400/70 hover:bg-[#191714]",
        dishId ? "scroll-mt-32" : "",
        highlighted ? "dish-highlight" : "",
      ].join(" ")}
    >`;

if (!menu.includes("function MenuCard({ item, dishId")) {
  if (!menu.includes(oldMenuCard)) {
    throw new Error("Could not find original MenuCard.");
  }

  menu = menu.replace(oldMenuCard, newMenuCard);
}

/* Replace FULL MENU card mapping */
const oldFullMap = `                  {section.items.map((item) => (
                    <MenuCard
                      key={item.name}
                      item={item}
                    />
                  ))}`;

const newFullMap = `                  {section.items.map((item) => {
                    const dishId = \`dish-\${item.name
                      .normalize("NFD")
                      .replace(/[\\\\u0300-\\\\u036f]/g, "")
                      .toLowerCase()
                      .replace(/[^a-z0-9]+/g, "-")
                      .replace(/(^-|-$)/g, "")}\`;

                    return (
                      <MenuCard
                        key={item.name}
                        item={item}
                        dishId={dishId}
                        highlighted={highlightedDish === dishId}
                      />
                    );
                  })}`;

if (!menu.includes("highlighted={highlightedDish === dishId}")) {
  if (!menu.includes(oldFullMap)) {
    throw new Error("Could not find full-menu card mapping.");
  }

  menu = menu.replace(oldFullMap, newFullMap);
}

/* Ensure Atlantic Salmon is using canonical image */
menu = menu.replace(
  /name:\s*"Atlantic Salmon",([\s\S]*?)image:\s*[\s\S]*?,\s*tag:\s*"Featured"/,
  (match, afterName) => {
    return `name: "Atlantic Salmon",${afterName}image:
          "/images/menu/seafood/atlantic-salmon.jpg",
        tag: "Featured"`;
  }
);

/* Ensure Crème Brûlée spelling is correct */
menu = menu.replace(
  /name:\s*"Cr[èe]me Br[ûu]l[ée]+e?"/g,
  'name: "Crème Brûlée"'
);

write(menuFile, menu);

/* =====================================================
   HOME FEATURED DISHES
===================================================== */

let featured = read(featuredFile);

const featuredLinks = {
  "Wagyu Ribeye": "/menu#dish-wagyu-ribeye",
  "Atlantic Salmon": "/menu#dish-atlantic-salmon",
  "Lobster Linguine": "/menu#dish-lobster-linguine",
};

for (const [dish, link] of Object.entries(featuredLinks)) {
  const escapedDish = dish.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

  const dishBlock = new RegExp(
    `(name:\\s*"${escapedDish}"[\\s\\S]*?)(to:\\s*["'][^"']+["'])`,
    "m"
  );

  if (dishBlock.test(featured)) {
    featured = featured.replace(
      dishBlock,
      `$1to: "${link}"`
    );
  }
}

featured = featured.replace(
  /to=["']\/menu#featured["']/g,
  'to="/menu#dish-wagyu-ribeye"'
);

write(featuredFile, featured);

/* =====================================================
   HOME CHEF SPECIALS
===================================================== */

let specials = read(specialsFile);

const specialsLinks = {
  "Herb Crusted Rack of Lamb":
    "/menu#dish-herb-crusted-rack-of-lamb",
  "Seafood Paella":
    "/menu#dish-seafood-paella",
  "Crème Brûlée":
    "/menu#dish-creme-brulee",
};

for (const [dish, link] of Object.entries(specialsLinks)) {
  const escapedDish = dish.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

  const dishBlock = new RegExp(
    `(name:\\s*"${escapedDish}"[\\s\\S]*?)(to:\\s*["'][^"']+["'])`,
    "m"
  );

  if (dishBlock.test(specials)) {
    specials = specials.replace(
      dishBlock,
      `$1to: "${link}"`
    );
  }
}

specials = specials.replace(
  /to=["']\/menu#specials["']/g,
  'to="/menu#dish-herb-crusted-rack-of-lamb"'
);

write(specialsFile, specials);

/* =====================================================
   GLOBAL GOLD HIGHLIGHT EFFECT
===================================================== */

let css = read(cssFile);

if (!css.includes(".dish-highlight")) {
  css += `

/* Ember & Oak exact dish navigation highlight */

@keyframes emberOakDishHighlight {
  0% {
    border-color: rgb(63 63 70);
    box-shadow: 0 0 0 rgba(251, 191, 36, 0);
  }

  15% {
    border-color: rgb(251 191 36);
    box-shadow:
      0 0 0 3px rgba(251, 191, 36, 0.18),
      0 0 32px rgba(251, 191, 36, 0.32);
  }

  70% {
    border-color: rgb(251 191 36);
    box-shadow:
      0 0 0 3px rgba(251, 191, 36, 0.18),
      0 0 32px rgba(251, 191, 36, 0.32);
  }

  100% {
    border-color: rgb(63 63 70);
    box-shadow: 0 0 0 rgba(251, 191, 36, 0);
  }
}

.dish-highlight {
  animation: emberOakDishHighlight 3s ease-in-out;
}
`;

  write(cssFile, css);
}

/* =====================================================
   FINAL VERIFICATION
===================================================== */

const checks = [
  ["MenuPage useEffect", menu.includes("useEffect")],
  ["MenuPage useLocation", menu.includes("useLocation")],
  ["highlightedDish state", menu.includes("highlightedDish")],
  ["dish IDs", menu.includes("const dishId = `dish-${item.name")],
  ["highlight class", menu.includes("dish-highlight")],
  ["Atlantic Salmon image", menu.includes("/images/menu/seafood/atlantic-salmon.jpg")],
  ["Crème Brûlée", menu.includes('name: "Crème Brûlée"')],
  ["Featured navigation", featured.includes("/menu#dish-")],
  ["Special navigation", specials.includes("/menu#dish-")],
  ["Global highlight CSS", css.includes(".dish-highlight")],
];

const failed = checks
  .filter(([, passed]) => !passed)
  .map(([name]) => name);

if (failed.length) {
  throw new Error(
    `Navigation fix incomplete. Failed checks:\n- ${failed.join("\n- ")}`
  );
}
